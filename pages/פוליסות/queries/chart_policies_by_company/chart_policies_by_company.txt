SELECT company, COUNT(*) AS count FROM gama_policies_with_partner WHERE (COALESCE(NULLIF({{company_filter.selectedOptionValue}}, ''), company) IS NOT DISTINCT FROM company) AND (COALESCE(NULLIF({{PartnerFilter.selectedOptionValue}}, ''), partner_name) IS NOT DISTINCT FROM partner_name) AND (COALESCE(NULLIF({{YearFilter.selectedOptionValue}}, ''), EXTRACT(YEAR FROM date)::text) IS NOT DISTINCT FROM EXTRACT(YEAR FROM date)::text) AND (COALESCE(NULLIF({{MonthFilter.selectedOptionValue}}, ''), EXTRACT(MONTH FROM date)::text) IS NOT DISTINCT FROM EXTRACT(MONTH FROM date)::text) GROUP BY company ORDER BY count DESC
