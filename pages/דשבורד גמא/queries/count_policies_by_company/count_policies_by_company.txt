SELECT company, COUNT(*) AS count FROM gama_policies WHERE (COALESCE(NULLIF({{MonthFilter.selectedOptionValue}}, '')::int, EXTRACT(MONTH FROM date)::int) IS NOT DISTINCT FROM EXTRACT(MONTH FROM date)::int) AND (COALESCE(NULLIF({{YearFilter.selectedOptionValue}}, '')::int, EXTRACT(YEAR FROM date)::int) IS NOT DISTINCT FROM EXTRACT(YEAR FROM date)::int) AND (COALESCE(NULLIF({{CompanyFilter.selectedOptionValue}}, ''), company) IS NOT DISTINCT FROM company) AND (COALESCE(NULLIF({{ActionFilter.selectedOptionValue}}, ''), action) IS NOT DISTINCT FROM action) GROUP BY company ORDER BY count DESC
