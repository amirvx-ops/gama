SELECT COUNT(*) AS total_meetings, COUNT(*) FILTER (WHERE meeting_status = 'התקיימה') AS completed, COUNT(*) FILTER (WHERE meeting_status = 'לא התקיימה') AS cancelled, COUNT(*) FILTER (WHERE meeting_status = 'בוטלה') AS cancelled2, COUNT(*) FILTER (WHERE meeting_status = 'נדחתה') AS postponed, COUNT(*) FILTER (WHERE meeting_status = 'אושרה') AS approved, CASE WHEN COUNT(*) FILTER (WHERE meeting_status IN ('התקיימה','לא התקיימה')) = 0 THEN 0 ELSE ROUND(100.0 * COUNT(*) FILTER (WHERE meeting_status = 'התקיימה') / COUNT(*) FILTER (WHERE meeting_status IN ('התקיימה','לא התקיימה')), 1) END AS pct_completed FROM gama_meetings_with_partner WHERE (COALESCE(NULLIF({{PartnerFilter.selectedOptionValue}}, ''), partner_name) IS NOT DISTINCT FROM partner_name) AND (COALESCE(NULLIF({{YearFilter.selectedOptionValue}}, ''), EXTRACT(YEAR FROM created_time)::text) IS NOT DISTINCT FROM EXTRACT(YEAR FROM created_time)::text) AND (COALESCE(NULLIF({{MonthFilter.selectedOptionValue}}, ''), EXTRACT(MONTH FROM created_time)::text) IS NOT DISTINCT FROM EXTRACT(MONTH FROM created_time)::text)
