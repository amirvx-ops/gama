SELECT id, event_type, record_id, TO_CHAR(created_time, 'DD/MM/YYYY') AS created_time, id_number, name, meeting_status, TO_CHAR(sent_at::timestamptz, 'DD/MM/YYYY HH24:MI') AS sent_at, partner_name, policy_closed FROM gama_meetings_with_partner WHERE (COALESCE({{Select_Status.selectedOptionValue}}, '') = '' OR meeting_status = {{Select_Status.selectedOptionValue}}) AND (COALESCE(NULLIF({{PartnerFilter.selectedOptionValue}}, ''), partner_name) IS NOT DISTINCT FROM partner_name) AND (COALESCE(NULLIF({{YearFilter.selectedOptionValue}}, ''), EXTRACT(YEAR FROM created_time)::text) IS NOT DISTINCT FROM EXTRACT(YEAR FROM created_time)::text) AND (COALESCE(NULLIF({{MonthFilter.selectedOptionValue}}, ''), EXTRACT(MONTH FROM created_time)::text) IS NOT DISTINCT FROM EXTRACT(MONTH FROM created_time)::text) ORDER BY created_time DESC
